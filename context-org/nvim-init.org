** folding
- https://github.com/neovim/neovim/blob/master/runtime/doc/fold.txt
使用 Treesitter 为支持的语言生成代码折叠层级，不额外安装折叠插件。

- =foldmethod= 使用表达式计算折叠层级。
- =foldexpr= 调用 Neovim 的 Treesitter 折叠表达式。
- =foldlevel= 和 =foldlevelstart= 设为 =99=，启动时默认展开代码。
- =foldenable= 启用折叠功能，=foldcolumn= 显示左侧折叠栏。
- 使用 =za= 切换、=zo= 打开、=zc= 关闭当前折叠，使用 =zR= 和 =zM= 操作全部折叠。
#+begin_src lua :tangle (concat lua-dir "options/init.lua") 
  -- 使用 Treesitter 计算代码折叠层级
  vim.opt.foldmethod = "expr"
  vim.opt.foldexpr = "v:lua.vim.treesitter.foldexpr()"
  -- 启动时展开全部折叠，但仍保留手动折叠能力
  vim.opt.foldlevel = 99
  vim.opt.foldlevelstart = 99
  vim.opt.foldenable = true
  vim.opt.foldcolumn = "1"
  #+end_src

*** config
#+begin_src lua :tangle (concat lua-dir "pkgs/auto-save.lua") 
  local gh = require("libs").gh

  vim.pack.add({
    {
      src = gh("okuuva/auto-save.nvim"),
      version = "v1.0.0",
    },
  })

  local excluded_filetypes = {
    "gitcommit",
    "help",
    "NvimTree",
    "prompt",
    "qf",
    "TelescopePrompt",
    "terminal",
    "toggleterm",
  }

  require("auto-save").setup({
    enabled = true,
    trigger_events = {
      immediate_save = { "BufLeave", "FocusLost", "QuitPre", "VimSuspend" },
      defer_save = { "InsertLeave", "TextChanged" },
      cancel_deferred_save = { "InsertEnter" },
    },
    condition = function(buf)
      if vim.fn.getbufvar(buf, "&buftype") ~= "" then
        return false
      end

      if vim.fn.getbufvar(buf, "&modifiable") ~= 1 then
        return false
      end

      if vim.fn.getbufvar(buf, "&readonly") == 1 then
        return false
      end

      if vim.api.nvim_buf_get_name(buf) == "" then
        return false
      end

      return not vim.tbl_contains(excluded_filetypes, vim.bo[buf].filetype)
    end,
    write_all_buffers = false,
    debounce_delay = 1000,
  })

#+end_src

** nvim-treesitter
*** docs
- https://github.com/nvim-treesitter/nvim-treesitter
- tree-sitter
  也需要提前安装，本项目配置再devenv环境中。.
  之后才可以编译Treesitter
- Treesitter parser 需要单独安装

新增语言时需要完成以下步骤：

1. 安装对应的 Treesitter parser。每个 parser 在当前 =NVIM_APPNAME= 数据目录下只需安装一次：

   #+begin_src sh
   nvim --headless \
     '+lua require("nvim-treesitter").install({"json"}):wait(300000)' \
     '+qa'
   #+end_src

2. 将对应的 filetype 添加到下面配置的 =FileType pattern= 列表中，使打开文件时启动 Treesitter。

3. 重新打开文件，并使用 =:set filetype?= 和 =:InspectTree= 检查 filetype 与语法树。

parser 名称和 filetype 名称不一定相同。例如 =typescript= 是 parser 名称，=typescriptreact= 是 React 文件的 filetype；添加语言时应分别确认两者。

当前配置采用手动安装 parser 的方式，避免 Neovim 启动时联网或编译。更换 =NVIM_APPNAME= 数据目录、清空 parser 目录或需要更新 parser 时，才需要重新执行安装命令。

  #+begin_src shell
    # - :TSInstall bash json jsonc lua markdown markdown_inline nix vim vimdoc query
    nvim --headless \
         '+lua require("nvim-treesitter").install({"bash", "json", "jsonc", "lua", "markdown", "markdown_inline", "nix", "vim", "vimdoc", "query"}):wait(300000)' \
         '+qa'
    #+end_src

  -- stylua: ignore start
  local reload =  function()
     local config = vim.env.MYVIMRC
     if not config or config == "" then
        config = vim.fn.stdpath("config") .. "/init.lua"
     end
     vim.cmd("source " .. vim.fn.fnameescape(config))
     vim.notify("Neovim configuration reloaded", vim.log.levels.INFO)
  end

  local triggle_tabline= function()
     vim.o.showtabline = vim.o.showtabline == 0 and 2 or 0
  end

  wk.add({
  	  { "<leader>w", "<cmd>write<cr>", desc = "Save file" },
  	  { "<leader>W", "<cmd>wall<cr>", desc = "Save all file" },
  	  { "<leader>q", "<cmd>quit<cr>", desc = "Quit window" },
  	  { "<leader>Q", "<cmd>quitall<cr>", desc = "Quit all window" },
  	  { "<leader>r",  reload, desc = "Reload Neovim configuration" },
  	  { "<leader>ua", "<cmd>ASToggle<cr>", desc = "Toggle auto-save" },
  	  { "jk", "<Esc>", mode = "i", desc = "Exit insert mode" },
  	  { "<leader>uB", triggle_tabline, desc = "Toggle bufferline" },
  	  { "<C-x><C-;>", "gcc", mode = "n", remap = true, desc = "Toggle comment" },
  	  { "<C-x><C-;>", "gc", mode = "v", remap = true, desc = "Toggle comment" },

  	    require("conform").format({ lsp_format = "fallback" })

     local is_empty_unnamed = vim.api.nvim_buf_get_name(0) == ""

** bufferline
*** docs
- https://github.com/akinsho/bufferline.nvim
*** file display
- =max_name_length= ：文件名最大显示长度。
- =max_prefix_length= ：同名文件时显示的路径前缀最大长度。
- =truncate_names= ：文件名过长时是否截断。
- =show_buffer_icons= ：是否显示文件类型图标。
- =show_buffer_close_icons= ：是否显示每个 Buffer 的关闭图标。
- =show_close_icon= ：是否显示 Bufferline 总体关闭图标。
- =show_tab_indicators= ：是否显示当前 Buffer 指示器。
- =show_duplicate_prefix= ：同名文件是否显示路径前缀。
- =modified_icon= ：文件修改状态图标。
- =name_formatter= ：自定义 Buffer 名称；当前配置隐藏最后一个文件后缀，并保留修改状态标记。
*** config
#+begin_src lua :tangle (concat lua-dir "pkgs/bufferline.lua") 
  local gh = require("libs").gh

  vim.pack.add({
  	  { src = gh("akinsho/bufferline.nvim") },
  	  { src = gh("nvim-tree/nvim-web-devicons") },
  })

  require("bufferline").setup({
  	  options = {
  		 mode = "buffers",
  		 indicator = {
  			style = "underline",
  		 },
  		 max_name_length = 20,
  		 max_prefix_length = 20,
  		 truncate_names = true,
  		 show_buffer_icons = true,
  		 show_buffer_close_icons = false,
  		 show_close_icon = false,
  		 show_tab_indicators = true,
  		 show_duplicate_prefix = true,
  		 modified_icon = "",
  		 --  在打开多个 Buffer 或切换 Tab 时会自动把 vim.o.showtabline = 2
  		 auto_toggle_bufferline = false,
  		 always_show_bufferline = false,
  		 themable = true,
  		 diagnostics = "nvim_lsp",
  		 separator_style = "none",
  		 show_buffer_close_icons = false,
  		 offsets = {
  			{
  			   filetype = "NvimTree",
  			   text = "",
  			   highlight = "Normal",
  			   text_align = "left",
  			   separator = false,
  			},
  		 },
  		 name_formatter = function(buf)
  			local name = vim.fn.fnamemodify(buf.name, ":t:r")
  			if vim.bo[buf.bufnr].modified then
  			   return name .. " ●"
  			end
  			return name
  		 end,
  	  },
  						   })

  -- 默认隐藏 Bufferline，由 Which-key 的 <leader>uB 手动切换。
  vim.o.showtabline = 0
