#+begin_src lua :tangle (concat lua-dir "options/init.lua") 
  vim.opt.number = true
  vim.opt.relativenumber = true
  vim.opt.mouse = "a"
  vim.opt.termguicolors = true
  vim.opt.expandtab = true
  vim.opt.shiftwidth = 2
  vim.opt.tabstop = 2
  vim.opt.signcolumn = "yes"
  vim.opt.updatetime = 250
#+end_src

** leader key
#+begin_src lua :tangle (concat lua-dir "keymap/init.lua") 
  vim.g.mapleader = " "
  vim.g.maplocalleader = " "
#+end_src
** shortcut
#+begin_src lua :tangle (concat lua-dir "keymap/init.lua") 
  vim.keymap.set("n", "<leader>w", "<cmd>write<cr>", { desc = "Save file" })
  vim.keymap.set("n", "<leader>q", "<cmd>quit<cr>", { desc = "Quit window" })
  vim.keymap.set("n", "<leader>r", function()
  				  local config = vim.env.MYVIMRC
  				  if not config or config == "" then
  					 config = vim.fn.stdpath("config") .. "/init.lua"
  				  end

  				  vim.cmd("source " .. vim.fn.fnameescape(config))
  				  vim.notify("Neovim configuration reloaded", vim.log.levels.INFO)
  end, { desc = "Reload Neovim configuration" })
#+end_src
